```python
"""
solution.py

Meeting slot recommendation module.

This module provides functionality to recommend N available meeting slots
within defined working hours while respecting existing busy intervals,
meeting duration requirements, optional buffers, and an optional candidate
time window.

Design Rules Implemented
------------------------
- Time representation: datetime.datetime with minute precision
- Interval type: List[Tuple[datetime, datetime]]
- Interval semantics: half-open intervals [start, end)
- Busy intervals are preprocessed (sorted and merged)
- Suggested slots are independent options (not sequential bookings)
- Buffer applies around suggested meeting slot
- Start times evaluated at 1-minute granularity
- Candidate window intersects with working hours
- Deterministic chronological output
"""

from datetime import datetime, timedelta
from typing import List, Tuple, Optional


Interval = Tuple[datetime, datetime]


def _validate_interval(interval: Interval) -> None:
    """Validate that an interval has start <= end."""
    start, end = interval
    if start > end:
        raise ValueError("Interval start time must not be after end time.")


def _merge_intervals(intervals: List[Interval]) -> List[Interval]:
    """
    Sort and merge overlapping or touching intervals.

    Intervals follow half-open semantics [start, end).
    Touching intervals (end == next_start) are merged.
    """
    if not intervals:
        return []

    intervals = sorted(intervals, key=lambda x: x[0])
    merged: List[Interval] = [intervals[0]]

    for start, end in intervals[1:]:
        last_start, last_end = merged[-1]

        if start <= last_end:  # overlap or touching
            merged[-1] = (last_start, max(last_end, end))
        else:
            merged.append((start, end))

    return merged


def _clip_to_window(intervals: List[Interval], window: Interval) -> List[Interval]:
    """Clip intervals so they only exist within the provided window."""
    window_start, window_end = window
    clipped: List[Interval] = []

    for start, end in intervals:
        if end <= window_start or start >= window_end:
            continue

        clipped_start = max(start, window_start)
        clipped_end = min(end, window_end)

        clipped.append((clipped_start, clipped_end))

    return clipped


def recommend_meeting_slots(
    working_hours: Interval,
    busy_intervals: List[Interval],
    duration_minutes: int,
    n: int,
    buffer_minutes: int = 0,
    candidate_window: Optional[Interval] = None,
) -> List[Interval]:
    """
    Recommend N available meeting slots.

    Parameters
    ----------
    working_hours : Tuple[datetime, datetime]
        The overall working hours window.

    busy_intervals : List[Tuple[datetime, datetime]]
        Existing busy intervals.

    duration_minutes : int
        Required meeting duration in minutes.

    n : int
        Number of meeting slots to return.

    buffer_minutes : int, optional
        Buffer before and after suggested meeting slots.

    candidate_window : Optional[Tuple[datetime, datetime]]
        Optional window restricting the search further.

    Returns
    -------
    List[Tuple[datetime, datetime]]
        Chronologically ordered meeting slot suggestions.
    """

    # ---------------------------
    # Validate inputs
    # ---------------------------

    if duration_minutes <= 0:
        raise ValueError("Meeting duration must be positive.")

    if n <= 0:
        raise ValueError("Number of requested slots must be positive.")

    _validate_interval(working_hours)

    if candidate_window:
        _validate_interval(candidate_window)

    for interval in busy_intervals:
        _validate_interval(interval)

    duration = timedelta(minutes=duration_minutes)
    buffer = timedelta(minutes=buffer_minutes)

    # ---------------------------
    # Determine effective window
    # ---------------------------

    window_start, window_end = working_hours

    if candidate_window:
        cand_start, cand_end = candidate_window
        window_start = max(window_start, cand_start)
        window_end = min(window_end, cand_end)

        if window_start >= window_end:
            return []

    search_window = (window_start, window_end)

    # ---------------------------
    # Preprocess busy intervals
    # ---------------------------

    clipped_busy = _clip_to_window(busy_intervals, search_window)
    merged_busy = _merge_intervals(clipped_busy)

    # ---------------------------
    # Slot search
    # ---------------------------

    results: List[Interval] = []
    current = window_start
    step = timedelta(minutes=1)

    while current + duration <= window_end and len(results) < n:

        slot_start = current
        slot_end = slot_start + duration

        candidate_valid = True

        for busy_start, busy_end in merged_busy:

            # Enforce buffers around meeting
            protected_start = slot_start - buffer
            protected_end = slot_end + buffer

            if not (protected_end <= busy_start or protected_start >= busy_end):
                candidate_valid = False
                break

        if candidate_valid:
            results.append((slot_start, slot_end))

        current += step

    return results
```

Example merge:


```
[start, end)
```

Meaning:

```
busy: 10:00–11:00
meeting may start exactly at 11:00
```

---

### ✔ Buffer Enforcement

Buffers are applied to the **candidate meeting slot**:

```
protected_start = slot_start - buffer
protected_end   = slot_end + buffer
```

The meeting is valid only if that protected interval **does not intersect busy intervals**.

### ✔ Candidate Window Intersection

```

    working_hours ∩ candidate_window
```

### ✔ Error Handling

* `duration_minutes <= 0`
* `n <= 0`
* `start > end` in any interval
